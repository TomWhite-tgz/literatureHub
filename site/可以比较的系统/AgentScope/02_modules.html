16_check_default（permission/_engine.py:117）

async def _check_default(self, tool, tool_input) -> PermissionDecision

permission py line _check_default

DEFAULT 模式的评估链，6 步顺序检查：1) Deny rules 命中则 DENY，最高优先级；2) Ask rules 命中则 ASK（带建议规则）；3) Read-only fast path 只读工具直接 ALLOW；4) Tool 自身 check_permissions：ALLOW/DENY 直接返回，安全 ASK（bypass-immune）带建议返回，非安全 ASK / PASSTHROUGH 继续；5) Allow rules 命中则 ALLOW；6) Mode fallback 默认 ASK（带建议规则）。

★ 安全 ASK 是 bypass-immune 的 安全 ASK（如 rm -rf /）即使在 BYPASS 模式下也不会被绕过——除非有显式 deny 规则。这是 AgentScope 权限模型的底线安全设计。DONT_ASK 模式更保守：把所有 ASK 都转成 DENY。

源码：

    async def _check_default(
        self,
        tool: ToolBase,
        tool_input: dict[str, Any],
    ) -> PermissionDecision:
        

        deny = await self._check_deny_rules(tool, tool_input)
        if deny:
            return deny

        ask = await self._check_ask_rules(tool, tool_input)
        if ask:
            ask.suggested_rules = await self._generate_suggestions(
                tool,
                tool_input,
            )
            return ask

        read_only = await self._check_read_only_fast_path(tool, tool_input)
        if read_only:
            return read_only

        tool_decision = await tool.check_permissions(tool_input, self.context)
        if tool_decision.behavior in (
            PermissionBehavior.ALLOW,
            PermissionBehavior.DENY,
        ):
            return tool_decision
        if self._is_safety_ask(tool_decision):
            tool_decision.suggested_rules = await self._generate_suggestions(
                tool,
                tool_input,
            )
            return tool_decision

        allow = await self._check_allow_rules(tool, tool_input)
        if allow:
            return allow

        default = PermissionDecision(
            behavior=PermissionBehavior.ASK,
            message=f"Permission required for {tool.name}",
            decision_reason=f"Mode: {self.context.mode.value}",
        )
        default.suggested_rules = await self._generate_suggestions(
            tool,
            tool_input,
        )
        return default

        self,
        *args: Any,
        **kwargs: Any,
    ) -> ToolChunk | AsyncGenerator[ToolChunk, None]:

            raise TypeError(
                f"{type(self).__name__} must be called with keyword arguments "
                f"only, but got {len(args)} positional argument(s).",
            )

        middlewares = getattr(self, "_middlewares", [])
        if not middlewares:
            if inspect.isasyncgenfunction(self.call):
                return self.call(**kwargs)
            return await self.call(**kwargs)

        async def execute_chain(
            index: int = 0,
            **chain_kwargs: Any,
        ) -> AsyncGenerator[ToolChunk, None]:
            
            if index >= len(middlewares):

                if inspect.isasyncgenfunction(self.call):
                    async for chunk in self.call(**chain_kwargs):
                        yield chunk
                else:
                    result = await self.call(**chain_kwargs)
                    if isinstance(result, AsyncGenerator):
                        async for chunk in result:
                            yield chunk
                    else:
                        yield result
            else:
                mw = middlewares[index]
                input_kwargs = dict(chain_kwargs)

                async def next_handler(
                    **kw: Any,
                ) -> AsyncGenerator[ToolChunk, None]:
                    async for chunk in execute_chain(index + 1, **kw):
                        yield chunk

                async for chunk in mw.on_tool_call(
                    tool=self,
                    input_kwargs=input_kwargs,
                    next_handler=next_handler,

    async def _offload_data_block(self, block: DataBlock) -> DataBlock:
        
        if not isinstance(block.source, Base64Source):
            return block

        backend = self.get_backend()
        hash_str = hashlib.sha256(block.source.data.encode()).hexdigest()
        ext = mimetypes.guess_extension(block.source.media_type) or ".bin"
        path = backend.join_path(self._data_dir, f"{hash_str}{ext}")

        if not await backend.file_exists(path):
            await backend.write_file(
                path,
                base64.b64decode(block.source.data),
            )

        return DataBlock(
            id=block.id,
            name=block.name,
            source=URLSource(
                url=AnyUrl(self._path_to_file_uri(path)),
                media_type=block.source.media_type,

        return skills
